01

개요

세 가지 기능을 하나의 피처로 묶어 구현한다. 모두 "옷의 소속 보관함을 변경한다"는 동일한 기반 위에서 동작한다.

📦
다른 보관함으로 이동
옷 상세 화면에서 보관함 목록을 열어 원하는 보관함으로 이동. 기존 updateStorage() 재사용.
🗑️
휴지통
조직마다 자동 생성되는 시스템 보관함. 옷을 이동해 임시 보관. 자동 삭제 없음.
🏷️
판매 보관함
조직마다 자동 생성되는 시스템 보관함. 판매 예정 옷을 모아두는 공간.
핵심 설계 원칙
  • 시스템 보관함은 storages.storage_type 컬럼으로 구분 — 'normal' / 'trash' / 'for_sale'
  • 조직 첫 생성 시 두 시스템 보관함 자동 생성 (기존 사용자는 앱 첫 실행 시 생성)
  • 시스템 보관함은 이름 변경·삭제 UI 비노출
  • 홈 화면 표시 여부는 user_settings에 저장, 프로필에서 토글

DB 변경

기존 테이블 최소 변경 + 신규 테이블 1개 추가.

storages 테이블 — storage_type 컬럼 추가
기존 보관함은 모두 storage_type = 'normal'이 되며, 시스템 보관함만 별도 타입을 가진다.
SQL
ALTER TABLE storages
  ADD COLUMN storage_type TEXT NOT NULL DEFAULT 'normal'
  CHECK (storage_type IN ('normal', 'trash', 'for_sale'));
storage_type 설명 삭제 허용 이름 변경
normal 일반 사용자 보관함 허용 허용
trash 휴지통 (시스템) 불가 불가
for_sale 판매 보관함 (시스템) 불가 불가
user_settings 테이블 — 신규
SQL
CREATE TABLE user_settings (
  user_id    UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  show_trash BOOLEAN NOT NULL DEFAULT true,
  show_sale  BOOLEAN NOT NULL DEFAULT true,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

ALTER TABLE user_settings ENABLE ROW LEVEL SECURITY;

CREATE POLICY "own_settings" ON user_settings
  USING  (user_id = auth.uid())
  WITH CHECK (user_id = auth.uid());

// 조직 생성 직후 or 앱 시작 시 호출
Future<void> ensureSystemStorages(String orgId) async {
  for (final type in ['trash', 'for_sale']) {
    final name = type == 'trash' ? '휴지통' : '판매 보관함';
    final existing = await _db
        .from('storages')
        .select('id')
        .eq('org_id', orgId)
        .eq('storage_type', type)
        .maybeSingle();
    if (existing == null) {
      await _db.from('storages').insert({
        'org_id': orgId,
        'name': name,
        'storage_type': type,
        'qr_code': '${orgId}_$type',
      });
    }
  }
}

// 시스템 보관함 조회
Future<StorageLocation?> fetchSystemStorage(String type) async {
  final orgId = await getOrCreateOrgId();
  final row = await _db
      .from('storages')
      .select('*, locations(id, name)')
      .eq('org_id', orgId)
      .eq('storage_type', type)
      .maybeSingle();
  return row == null ? null : StorageLocation.fromMap(row);
}
UserSettingsRepository — 신규
Dart — lib/data/repositories/user_settings_repository.dart
class UserSettingsRepository {
  final SupabaseClient _db;
  UserSettingsRepository({SupabaseClient? db})
      : _db = db ?? Supabase.instance.client;

  Future<({bool showTrash, bool showSale})> fetchSettings() async {
    final userId = _db.auth.currentUser!.id;
    final row = await _db
        .from('user_settings')
        .select('show_trash, show_sale')
        .eq('user_id', userId)
        .maybeSingle();
    return (
      showTrash: row?['show_trash'] as bool? ?? true,
      showSale:  row?['show_sale']  as bool? ?? true,
    );
  }

  Future<void> updateShowTrash(bool value) => _upsert({'show_trash': value});
  Future<void> updateShowSale(bool value)  => _upsert({'show_sale': value});

  Future<void> _upsert(Map<String, dynamic> data) async {
    final userId = _db.auth.currentUser!.id;
    await _db.from('user_settings').upsert({
      'user_id': userId,
      'updated_at': DateTime.now().toIso8601String(),
      ...data,
    }, onConflict: 'user_id');
  }
}
