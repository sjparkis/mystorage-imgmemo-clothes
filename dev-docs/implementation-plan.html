create or replace function is_org_member(p_org_id uuid)
returns boolean language sql security definer as $$
  select exists (
    select 1 from organization_members
    where org_id = p_org_id and user_id = auth.uid()
  );
$$;

create policy "storages_all" on storages
  for all using (is_org_member(org_id));

create policy "clothes_all" on clothes
  for all using (is_org_member(org_id));

import Anthropic from "https://esm.sh/@anthropic-ai/sdk@0.27.3";

const anthropic = new Anthropic({ apiKey: Deno.env.get("ANTHROPIC_API_KEY")! });

serve(async (req) => {
  const { base64Image, contentType, imageUrl } = await req.json();
  const imageSource = base64Image
    ? { type: "base64" as const, media_type: contentType ?? "image/jpeg", data: base64Image }
    : { type: "url" as const, url: imageUrl };
  const response = await anthropic.messages.create({
    model: "claude-haiku-4-5-20251001",
    max_tokens: 300,
    messages: [{ role: "user", content: [
      { type: "image", source: imageSource },
      { type: "text", text: "의류 이미지를 분석해 JSON으로 응답: {\"name\":\"string\",\"tags\":[{\"name\":\"string\",\"category\":\"color|type|material|season|style|condition|brand|custom\"}]} 태그 7-10개"}
    ]}],
  });
  const text = response.content[0].type === "text" ? response.content[0].text : "{}";
  return new Response(text, {
    headers: { "Content-Type": "application/json" }
  });
});

class AiTagService {
  Future<({String name, List<Tag> tags})> analyze({
    String? base64Image, String? contentType, String? imageUrl,
  }) async {
    final body = base64Image != null
      ? {'base64Image': base64Image, 'contentType': contentType ?? 'image/jpeg'}
      : {'imageUrl': imageUrl};
    final res = await Supabase.instance.client.functions.invoke(
      'analyze-clothing', body: body,
    );
    final tags = (res.data['tags'] as List).map((t) => Tag(
      id: '', name: t['name'], category: t['category'], aiGenerated: true,
    )).toList();
    return (name: res.data['name'] as String, tags: tags);
  }
}

class VoiceSearchService {
  final _stt = SpeechToText();

  Future<String?> listen() async {
    await _stt.initialize();
    String? result;
    await _stt.listen(
      onResult: (r) { if (r.finalResult) result = r.recognizedWords; },
      listenFor: const Duration(seconds: 10),
      localeId: 'ko_KR',
    );
    return result;
  }

  Future<Map<String, dynamic>> parseQuery(String text) async {
    final res = await Supabase.instance.client.functions
        .invoke('parse-voice-query', body: {'query': text});
    return res.data as Map<String, dynamic>;
  }
}

serve(async (req) => {
  const { event, app_user_id, subscriber } = await req.json();
  const entitlements = subscriber?.entitlements ?? {};
  let plan = "free";
  if (entitlements["ai_voice"]?.is_active) plan = "ai_voice";
  else if (entitlements["basic"]?.is_active) plan = "basic";
  else if (entitlements["ad_free"]?.is_active) plan = "ad_free";

  await supabase.from("subscriptions").upsert(
    { user_id: app_user_id, plan, status: plan === "free" ? "expired" : "active" },
    { onConflict: "user_id" }
  );
  return new Response("ok");
});

SubscriptionService — 플랜 조회 + 한도 체크 + 구매
class SubscriptionService {
  Future<bool> canAddStorage(int current) async {
    final plan = await getCurrentPlan();
    final limit = AppConstants.planLimits[plan]!['storages']!;
    return limit == -1 || current < limit;
  }

  Future<bool> hasVoiceSearch() async =>
      (await getCurrentPlan()) == 'ai_voice';

  Future<void> purchase(String productId) async {
    final offerings = await Purchases.getOfferings();
    final pkg = offerings.current?.availablePackages
        .firstWhere((p) => p.storeProduct.identifier == productId);
    if (pkg != null) await Purchases.purchasePackage(pkg);
  }

핵심 제약사항

Apple 인앱결제 강제 & 외부 결제 유도 금지
iOS 앱 내 디지털 상품 결제는 반드시 Apple In-App Purchase 사용 필수. 앱 내에서 "웹에서 더 저렴하게 구매" 같은 외부 결제 유도 문구는 앱 심사 거절 사유. iOS · Android · Web 모두 동일 가격 표시가 원칙.
수수료: App Store 30% (구독 1년 이후 15%) / Google Play 15%
SQLite(Drift) 오프라인 캐시는 Web에서 작동 안 함
Flutter Web은 SQLite를 기본 지원하지 않음. 현재 설계의 Drift 오프라인 레이어는 iOS/Android 전용으로 처리.
Web PWA는 오프라인 지원 제외 (서비스 워커로 앱 쉘만 캐시). 웹 오프라인이 필요하면 별도로 sqlite3_web(WASM) 도입 검토.
Supabase Free 플랜 — 7일 비활성 시 프로젝트 자동 정지
1주일간 요청이 없으면 프로젝트가 일시정지됨. 베타 운영 중 갑자기 서비스가 멈출 수 있음.
유료 사용자가 생기면 즉시 Pro($25/월) 전환 필요. Pro 전환 시 정지 없음.
Flutter Web 초기 로딩 — 번들 크기 주의
Flutter Web(CanvasKit 렌더러) 초기 번들 약 2~5MB. 첫 방문 시 3G 환경에서 5~10초 소요.
PWA 설치 후 재방문은 빠름. --web-renderer html 옵션으로 번들 크기 절반 가능하나 렌더링 품질 저하.
Admin Web — 별도 Vercel 프로젝트로 분리 권장
일반 사용자 앱과 Admin을 같은 빌드에 포함하면 Admin 코드가 클라이언트에 노출됨.
Admin 전용 Flutter 빌드 타겟을 별도 Vercel 프로젝트로 배포하고, Supabase RLS + Service Role Key로 접근 제한.
